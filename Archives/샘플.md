```python
import json
from pathlib import Path
from pprint import pprint

from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import JsonOutputParser
from langchain_community.document_loaders import JSONLoader
from pydantic import BaseModel, Field

from dotenv import load_dotenv

load_dotenv()

# data = json.loads(Path("./files/words.json").read_text(encoding="utf-8"))


class Word(BaseModel):
    Frequency: str = Field(description="Rank (based on frequency of use)")
    Expression: str = Field(description="Original Japanese text")
    Reading: str = Field(
        description="Extract the exact hiragana reading of 'Expression' from Example_jp (e.g. for '彼の声はとても大きい' → 'こえ')"
    )
    Meaning: str = Field(
        description="Korean translation (multiple meanings separated by commas)"
    )
    Example_jp: str = Field(
        description="Japanese example sentence demonstrating the specified reading"
    )
    Example_kr: str = Field(description="Korean translation of the example sentence")


llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

loader = JSONLoader(
    file_path="./files/words.json",
    jq_schema=".[]",
    text_content=False,
)

documents = loader.load()

# 프롬프트 템플릿
template = """Complete this JSON file using the provided format.

Format rules:
{format_instructions}

Reference JSON structure:
{reference_json}
"""

prompt = ChatPromptTemplate.from_messages(
    [
        ("system", template),
    ]
)

# Parser 설정
parser = JsonOutputParser(pydantic_object=Word)
prompt = prompt.partial(format_instructions=parser.get_format_instructions())

# Chain 구성
chain = prompt | llm | parser

answer = chain.invoke({"reference_json": documents[13]})
print(answer)


```
